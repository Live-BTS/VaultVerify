    63→async function parseWorkbook(file: File): Promise<{ rows: ImportRow[]; sheetName: string; headerFound: boolean }> {
    64→  const buf = await file.arrayBuffer();
    65→  const wb = XLSX.read(buf, { type: "array" });
    66→  const sheetName = wb.SheetNames.find((n) => /skill/i.test(n)) ?? wb.SheetNames[0];
    67→  const ws = wb.Sheets[sheetName];
    68→  const aoa: string[][] = XLSX.utils.sheet_to_json(ws, { header: 1, blankrows: false, defval: "" });
    69→  // locate header row within the first 12 rows
    70→  let headerIdx = -1;
    71→  for (let i = 0; i < Math.min(aoa.length, 12); i++) {
    72→    const cells = (aoa[i] ?? []).map((c) => String(c ?? "").trim().toLowerCase());
    73→    if (cells.includes("skill name") && cells.includes("profession")) { headerIdx = i; break; }
    74→  }
    75→  if (headerIdx === -1) return { rows: [], sheetName, headerFound: false };
    76→  const headers = (aoa[headerIdx] ?? []).map((c) => String(c ?? "").trim());
    77→  const idx = (name: string) => headers.findIndex((h) => h.toLowerCase() === name.toLowerCase());
    78→  const colMap = { profession: idx("Profession"), jobTitle: idx("Job Title"), specialty: idx("Specialty"), category: idx("Category"), skillName: idx("Skill Name"), questionType: idx("Question Type"), hasNA: idx("Has N/A Option") };
    79→  const rows: ImportRow[] = [];
    80→  for (const cells of aoa.slice(headerIdx + 1)) {
    81→    const get = (i: number) => (i >= 0 ? String(cells[i] ?? "").trim() : "");
    82→    const r = { profession: get(colMap.profession), jobTitle: get(colMap.jobTitle), specialty: get(colMap.specialty), category: get(colMap.category), skillName: get(colMap.skillName), questionType: get(colMap.questionType), hasNA: get(colMap.hasNA).toLowerCase() === "yes" };
    83→    if (!r.profession && !r.jobTitle && !r.specialty && !r.category && !r.skillName) continue; // skip fully blank rows
    84→    rows.push(r);
    85→  }
    86→  return { rows, sheetName, headerFound: true };
    87→}

   101→export function SuperAdmin({ onExit }: { onExit: () => void }) {
   102→  const { toast } = useToast();
   103→  const [code, setCode] = useState("");
   104→  const [data, setData] = useState<Overview | null>(null);

   112→

   114→    const res = await fetch("/api/superadmin", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ code, ...payload }) });
   115→    const d = await res.json();
   116→    if (!res.ok || !d.ok) throw new Error(d.error ?? "Request failed");
   117→    return d;
   118→  }, [code]);
   119→
   120→  const auth = async (c?: string) => {
   121→    const useCode = c ?? code;
   122→    if (!useCode.trim()) return toast({ title: "Enter the superadmin code.", variant: "destructive" });
   123→    setLoading(true);
   124→    try {
   125→      const res = await fetch("/api/superadmin", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "auth", code: useCode }) });
   126→      const d = await res.json();
   127→      if (!res.ok || !d.ok) throw new Error(d.error ?? "Invalid code");
   128→      setCode(useCode);
   129→      setData(d);
   130→    } catch (e) {
   131→      toast({ title: e instanceof Error ? e.message : "Sign-in failed", variant: "destructive" });
   132→    } finally {
   133→      setLoading(false);
   134→    }
   135→  };
   136→
   137→  const refresh = async () => {
   138→    try { setData(await call({ action: "auth" })); } catch { /* keep old data */ }
   139→  };
   140→
   141→  const loadRequests = async () => {
   142→    try { const d = await call({ action: "requests" }); setCqRequests(d.requests); } catch { /* keep old */ }
   143→  };
   144→
   145→  const decide = async (id: string, approve: boolean) => {
   146→    try {
   147→      await call({ action: "decide", id, approve });
   148→      toast({ title: approve ? "Request approved" : "Request declined", description: approve ? "The candidate can now complete the checklist." : "The candidate will see the decision in their portal." });
   149→      await Promise.all([refresh(), loadRequests()]);
   150→    } catch { toast({ title: "Decision failed", variant: "destructive" }); }
   151→  };
   152→
   153→  const onFile = async (f: File | null) => {
   154→    if (!f) return;
   155→    try {
   156→      const { rows, sheetName, headerFound } = await parseWorkbook(f);
   157→      if (!headerFound) {
   158→        setPending(null);
   159→        return toast({ title: "Headers not found", description: `Sheet "${sheetName}" must contain the 7 MyZipVault columns (Profession … Has N/A Option).`, variant: "destructive" });
   160→      }
   161→      const { errors, clean } = validateRows(rows);
   162→      setPending({ rows, clean, errors, fileName: f.name });
   163→    } catch {
   164→      toast({ title: "Could not read that file", description: "Save the workbook as .xlsx and try again.", variant: "destructive" });
   165→    }
   166→  };
   167→
   168→  const runImport = async () => {
   169→    if (!pending) return;
   170→    setImporting(true);
   171→    try {
   172→      const d = await call({ action: "import", rows: pending.rows });
   173→      toast({ title: `Imported ${d.created} new · updated ${d.updated}`, description: d.errors?.length ? `${d.errors.length} row(s) rejected — see report.` : "All rows passed validation." });
   174→      setPending(null);
   175→      if (fileRef.current) fileRef.current.value = "";
   176→      await refresh();
   177→    } catch (e) {
   178→      toast({ title: e instanceof Error ? e.message : "Import failed", variant: "destructive" });
   179→    } finally {
   180→      setImporting(false);
   181→    }
   182→  };
   183→
   184→  const toggleRow = async (row: TemplateRow) => {
   185→    try { await call({ action: "toggle", id: row.id, active: !row.active }); await refresh(); } catch { toast({ title: "Update failed", variant: "destructive" }); }
   186→  };
   187→
   188→  const deleteSet = async (s: TemplateSet) => {
   189→    try {
   190→      const d = await call({ action: "deleteSet", profession: s.profession, jobTitle: s.jobTitle, specialty: s.specialty });

   192→      await refresh();
   193→    } catch { toast({ title: "Delete failed", variant: "destructive" }); }
   194→  };
   195→
   196→  // ── Gate ──
   197→  if (!data) {

   201→          <div className="flex flex-col items-center text-center">

   204→            <p className="mt-1 text-sm text-jade-muted">Platform control: users, companies, candidate data, credits, skills.</p>
   205→          </div>
   206→          <div className="mt-6">
   207→            <Label htmlFor="sa-code" className="text-jade-ink/80">Access code</Label>

   211→              value={code}
   212→              onChange={(e) => setCode(e.target.value)}
   213→              onKeyDown={(e) => e.key === "Enter" && auth()}
   214→              placeholder="Superadmin code"
   215→              className="mt-1.5 border-vault-border bg-white text-jade-ink placeholder:text-[#8aa29c]"
   216→            />
   217→            <p className="mt-2 text-xs text-jade-muted">
   218→              Sandbox demo code:{" "}
   219→              <button type="button" className="font-mono font-semibold text-verify-ink underline" onClick={() => auth("zipvault2026")}>
   220→                zipvault2026
   221→              </button>
   222→            </p>
   223→          </div>
   224→          <Button onClick={() => auth()} disabled={loading} className="mt-5 w-full bg-verify-green text-vault-dark hover:bg-verify-green/90">
   225→            {loading ? "Checking…" : "Unlock console"} <ShieldCheck className="ml-2 h-4 w-4" />
   226→          </Button>

   233→  }
   234→
   235→  const s = data.stats;

   249→  const navList = (

   252→        const active = section === sec;
   253→        const badge = badgeFor(sec);
   254→        return (

   256→            onClick={() => { setSection(sec); if (sec === "requests" && !cqRequests) loadRequests(); }}
   257→            className={cn(
   258→              "flex w-full items-center gap-2.5 rounded-lg px-3 py-2 text-sm font-medium transition",
   259→              active ? "bg-verify-green/15 text-verify-ink" : "text-jade-muted hover:bg-jade-ink/5 hover:text-jade-ink"
   260→            )}>
   261→            <Icon className={cn("h-4 w-4 shrink-0", active && "text-verify-ink")} />
   262→            <span className="flex-1 text-left">{label}</span>
   263→            {badge > 0 && <span className="rounded-full bg-amber-100 px-1.5 text-[10px] font-bold text-amber-700">{badge}</span>}
   264→          </button>
   265→        );
   266→      })}
   267→    </nav>
   268→  );
   269→
   270→  return (
   271→    <div className="vv-page min-h-screen">
   272→      {/* Header */}
   273→      <header className="sticky top-0 z-40 border-b border-vault-border/70 bg-white/85 backdrop-blur-xl">
   274→        <div className="mx-auto flex h-16 max-w-7xl items-center justify-between px-4 sm:px-6">
   275→          <div className="flex items-center gap-3">
   276→            <VaultMark size={32} />
   277→            <div className="leading-tight">
   278→              <p className="text-sm font-semibold text-jade-ink">
   279→                Vault<span className="text-verify-ink">Verify</span> <span className="ml-1 rounded bg-verify-green/15 px-1.5 py-0.5 text-[10px] font-bold uppercase tracking-wider text-verify-ink">Super Admin</span>
   280→              </p>
   281→              <p className="text-[11px] text-jade-muted">Zipvault skills platform console</p>
   282→            </div>
   283→          </div>
   284→          <div className="flex items-center gap-2">
   285→            <Button variant="ghost" onClick={refresh} className="px-3 text-[13px] text-jade-muted hover:text-jade-ink">
   286→              <RefreshCw className="h-3.5 w-3.5" /> Refresh
   287→            </Button>
   288→            <Button variant="ghost" onClick={onExit} className="px-3 text-[13px] text-jade-muted hover:text-jade-ink">Sign out</Button>
   289→          </div>

   294→        {/* Sidebar (desktop) */}

   306→          {/* Section switcher (mobile) */}
