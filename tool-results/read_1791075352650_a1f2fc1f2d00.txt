    70→export function ReferencesPanel({ email, fallbackName, onLaunchSetup, onStats, onShared }: {
    71→  email: string;
    72→  fallbackName: string;
    73→  onLaunchSetup: () => void;
    74→  onStats?: (s: { total: number; completed: number } | null) => void;
    75→  onShared?: () => void; // notify Controlled Sharing that a new reference link exists
    76→}) {
    77→  const { toast } = useToast();
    78→  const [candidate, setCandidate] = useState<RefCandidate | null>(null);
    79→  const [state, setState] = useState<"loading" | "none" | "ready" | "error">("loading");
    80→  const [error, setError] = useState<string | null>(null);
    81→  const [openReport, setOpenReport] = useState<string | null>(null);
    82→
    83→  const load = useCallback(async () => {
    84→    setState("loading");
    85→    setError(null);
    86→    try {
    87→      const res = await fetch(`/api/candidate?email=${encodeURIComponent(email)}`);
    88→      const d = await res.json();
    89→      if (res.status === 404) {
    90→        setCandidate(null);
    91→        setState("none");
    92→        onStats?.(null);
    93→        return;
    94→      }
    95→      if (!res.ok) throw new Error(d.error ?? "Could not load references");
    96→      const c: RefCandidate = d.candidate;
    97→      setCandidate(c);
    98→      setState("ready");
    99→      onStats?.({ total: c.requests.length, completed: c.requests.filter((r) => r.status === "COMPLETED" || r.status === "FLAGGED").length });
   100→    } catch (e) {
   101→      setError(e instanceof Error ? e.message : "Could not load references");
   102→      setState("error");
   103→    }
   104→  }, [email, onStats]);
   105→
   106→  useEffect(() => { load(); }, [load]);
   107→
   108→  const nudge = async (requestId: string, refName: string) => {
   109→    const res = await fetch("/api/candidate/actions", {
   110→      method: "POST", headers: { "Content-Type": "application/json" },
   111→      body: JSON.stringify({ action: "nudge", requestId }),
   112→    });
   113→    const d = await res.json();

   115→    else toast({ title: d.error ?? "Failed to send nudge", variant: "destructive" });
   116→  };

   119→    const res = await fetch("/api/candidate/actions", {
   120→      method: "POST", headers: { "Content-Type": "application/json" },
   121→      body: JSON.stringify({ action: "swap", requestId, replacement }),
   122→    });
   123→    const d = await res.json();
   124→    if (res.ok) {
   125→      toast({ title: d.message ?? "Replacement request sent" });
   126→      load();
   127→    } else {
   128→      toast({ title: d.error ?? "Swap failed", variant: "destructive" });
   129→    }
   130→  };
   131→
   132→  const addRef = async (reference: Record<string, string>) => {
   133→    const res = await fetch("/api/candidate/actions", {
   134→      method: "POST", headers: { "Content-Type": "application/json" },
   135→      body: JSON.stringify({ action: "add_reference", email, reference }),
   136→    });
   137→    const d = await res.json();
   138→    if (res.ok) {
   139→      toast({ title: d.message ?? "Request sent" });
   140→      load();
   141→    } else {
   142→      toast({ title: d.error ?? "Could not add reference", variant: "destructive" });
   143→    }
   144→  };
   145→
   146→  if (state === "loading") {
   147→    return <div className="flex justify-center py-16"><Spinner label="Loading your references…" /></div>;
   148→  }
   149→
   150→  if (state === "error") {
   151→    return (
   152→      <div className="mt-6 rounded-2xl border border-dashed border-vault-border bg-white/60 p-10 text-center">
   153→        <p className="text-sm font-medium text-rose-600">{error}</p>
   154→        <Button size="sm" variant="outline" onClick={load} className="mt-4">Try again</Button>
   155→      </div>
   156→    );
   157→  }
   158→
   159→  if (state === "none") {
   160→    return (
   161→      <div className="mt-6 rounded-2xl border border-dashed border-vault-border bg-white/60 p-10 text-center">
   162→        <Users className="mx-auto h-8 w-8 text-verify-ink/60" />
   163→        <p className="mt-3 text-sm font-semibold text-jade-ink">No reference profile yet</p>
   164→        <p className="mx-auto mt-1 max-w-md text-sm text-jade-muted">
   165→          VaultVerify verifies your experience with two professional references. Launch setup, add your references, and each one
   166→          receives a secure SMS + email link — they sign off within 14 days and the verified results land right here in your vault.
   167→        </p>
   168→        <div className="mx-auto mt-5 grid max-w-lg gap-2 text-left text-xs text-jade-muted sm:grid-cols-3">
   169→          {[
   170→            ["1", "Add 2 references", "Supervisor or charge nurse who saw your practice firsthand"],
   171→            ["2", "They sign securely", "Identity-checked form with a signature and audit trail"],
   172→            ["3", "Results land here", "Verified references live next to your skill checklists"],
   173→          ].map(([n, t, d]) => (
   174→            <div key={n} className="rounded-xl border border-vault-border bg-white p-3">
   175→              <p className="flex items-center gap-1.5 font-semibold text-verify-ink"><span className="flex h-4 w-4 items-center justify-center rounded-full bg-verify-green/20 text-[10px] font-bold">{n}</span> {t}</p>
   176→              <p className="mt-1 leading-relaxed">{d}</p>
   177→            </div>
   178→          ))}
   179→        </div>
   180→        <Button onClick={onLaunchSetup} className="mt-6 bg-verify-green text-vault-dark hover:bg-verify-green/90">
   181→          <UserRound className="mr-1.5 h-4 w-4" /> Set up references
   182→        </Button>
   183→        <p className="mt-3 text-[11px] text-[#8aa29c]">Setup pre-fills your details from this account ({email}).</p>
   184→      </div>

   186→  }

   189→
   190→  const parseAnswers = (raw: string): RefAnswer[] => { try { return JSON.parse(raw || "[]"); } catch { return []; } };
   191→  const fmtValue = (a: RefAnswer): string => {
   192→    if (a.unableToObserve) return "Unable to observe";
   193→    if (a.type === "rating" && typeof a.value === "number") return `${a.value} / 5`;
   194→    if (a.type === "boolean") return a.value === "YES" ? "Yes" : a.value === "NO" ? "No" : "Prefer not to say";

   196→  };
   197→  const valueClass = (a: RefAnswer): string => {
   198→    const v = fmtValue(a);
   199→    if (v.startsWith("5") || v === "Yes") return "border-verify-green/40 bg-verify-green/10 text-verify-ink";
   200→    if (v.startsWith("1") || v.startsWith("2") || v === "No") return "border-rose-200 bg-rose-50 text-rose-600";
   201→    if (v.startsWith("3") || v.startsWith("4")) return "border-[#5f87ae]/40 bg-[#5f87ae]/10 text-[#123c54]";
   202→    return "border-vault-border bg-jade-ink/5 text-jade-muted";

   204→
   205→  return (
   206→    <div className="mt-6 space-y-4">
   207→      {/* profile strip */}
   208→      <div className="flex flex-wrap items-center justify-between gap-3 rounded-2xl border border-vault-border bg-white vv-card-shadow p-5">
   209→        <div>
   210→          <p className="text-base font-semibold text-jade-ink">
   211→            {candidate!.fullName} <span className="font-normal text-jade-muted">· {candidate!.role}{candidate!.specialty ? `, ${specialtyLabel(candidate!.specialty)}` : ""}</span>
   212→          </p>
   213→          <p className="mt-1 text-xs text-jade-muted">
   214→            {candidate!.yearsExperience} yrs experience{candidate!.city ? ` · ${candidate!.city}, ${candidate!.state}` : ""} · {done}/{candidate!.requests.length} references completed
   215→          </p>
   216→        </div>
   217→        <span className="rounded-lg border border-verify-green/30 bg-verify-green/10 px-3 py-1.5 text-[11px] font-medium text-verify-ink">
   218→          Auto-reminders at day 2 · 5 · 9
   219→        </span>
   220→      </div>
   221→
   222→      <div className="flex justify-end">
   223→        <AddReferenceDialog onAdd={addRef} />
   224→      </div>
   225→
   226→      {/* request cards */}
   227→      {candidate!.requests.map((r) => {
   228→        const open = daysSince(r.sentAt);
   229→        const isDone = r.status === "COMPLETED" || r.status === "FLAGGED";
   230→        return (
   231→          <motion.div key={r.id} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.35 }}
   232→            className="rounded-2xl border border-vault-border bg-white vv-card-shadow p-5">
   233→            <div className="flex flex-wrap items-start justify-between gap-3">
   234→              <div>
   235→                <p className="font-semibold text-jade-ink">{r.refName} <span className="font-normal text-jade-muted">· {r.refTitle}</span></p>
   236→                <p className="mt-0.5 text-sm text-jade-muted">{r.facilityName} · {r.refEmail}{r.relationship ? ` · ${r.relationship}` : ""}</p>
   237→              </div>
   238→              <div className="flex items-center gap-2">
   239→                <StatusBadge status={r.status} />
   240→                {!isDone && <span className="rounded-full border border-vault-border px-2.5 py-0.5 text-xs text-jade-muted">Day {open}</span>}
   241→              </div>
   242→            </div>
   243→
   244→            {isDone && r.response && (
   245→              <div className="mt-3">
   246→                <p className="text-sm text-jade-ink">
   247→                  Completed{r.response.overallRating != null && <> — average rating <strong>{r.response.overallRating.toFixed(1)}/5</strong></>}
   248→                  {r.response.skillsVerified && <span className="ml-2 rounded-full border border-verify-green/40 bg-verify-green/10 px-2 py-0.5 text-[11px] font-semibold text-verify-ink">Skills verified</span>}
   249→                  {r.status === "FLAGGED" && <span className="ml-1 text-rose-600">(under review)</span>}
   250→                  <span className="block text-xs text-jade-muted">Signed {r.completedAt ? fmt(r.completedAt) : ""} · view it, download the PDF, or share it with anyone below.</span>
   251→                </p>
   252→                <div className="mt-3 flex flex-wrap items-center gap-2">
   253→                  <Button size="sm" variant="outline" className="border-vault-border text-jade-ink hover:bg-verify-green/10"
   254→                    onClick={() => setOpenReport(openReport === r.id ? null : r.id)}>
   255→                    {openReport === r.id ? <EyeOff className="mr-1.5 h-3.5 w-3.5" /> : <Eye className="mr-1.5 h-3.5 w-3.5" />}

   262→                  </a>
   263→                  <RefShareDialog request={{ id: r.id, refName: r.refName }} onCreated={() => { onShared?.(); load(); }} />
   264→                </div>
   265→                {openReport === r.id && (
   266→                  <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}
   267→                    className="mt-4 space-y-4 rounded-xl border border-vault-border bg-[#f7fbf8] p-4">
   268→                    <div>
   269→                      <p className="text-[11px] font-bold uppercase tracking-[0.14em] text-verify-ink/90">Verification answers</p>
   270→                      <div className="mt-2 divide-y divide-vault-border/60">

   273→                          return (
   274→                            <div key={a.key} className="flex flex-wrap items-baseline justify-between gap-x-6 gap-y-1 py-2">
   275→                              <p className="min-w-0 flex-1 text-xs text-jade-ink"><span className="mr-1 text-jade-muted">{i + 1}.</span>{q?.title ?? a.key}</p>
   276→                              <span className={cn("rounded-full border px-2.5 py-0.5 text-[11px] font-semibold", valueClass(a))}>{fmtValue(a)}</span>
   277→                            </div>
   278→                          );
   279→                        })}
   280→                      </div>
   281→                    </div>
   282→                    {r.response.remarks && (
   283→                      <div className="rounded-lg border border-vault-border bg-white p-3">
   284→                        <p className="text-[11px] font-bold uppercase tracking-wider text-verify-ink/90">Reference remarks</p>
   285→                        <p className="mt-1 whitespace-pre-wrap text-xs leading-relaxed text-jade-ink">{r.response.remarks}</p>
   286→                      </div>
   287→                    )}
   288→                    <div className="flex flex-wrap items-end justify-between gap-2 border-t border-vault-border/60 pt-3">
   289→                      <div>
   290→                        <p className="font-serif text-lg italic text-jade-ink">{r.response.signatureName}</p>
   291→                        <p className="text-[11px] text-jade-muted">{r.refName}, {r.refTitle} · signed {r.response.signedAt ? fmt(r.response.signedAt) : ""}</p>
   292→                      </div>
   293→                      <p className="rounded-md bg-[#f0f6f2] px-2.5 py-1 text-[10px] font-medium text-[#33565b]"><ShieldCheck className="mr-1 inline h-3 w-3" /> Typed e-signature · audit-trailed</p>
   294→                    </div>
   295→                  </motion.div>
   296→                )}
   297→              </div>
   298→            )}
   299→
   300→            {!isDone && open >= 10 && (
   301→              <p className="mt-3 rounded-xl border border-amber-200 bg-amber-50 p-3 text-sm text-amber-800">
   302→                It&apos;s been {open} days. Consider nudging {r.refName.split(" ")[0]} or swapping in a different reference.
   303→              </p>
   304→            )}
   305→
   306→            <div className="mt-4 flex flex-wrap items-center gap-2">
   307→              {!isDone && (
   308→                <>
   309→                  <Button size="sm" variant="outline" onClick={() => nudge(r.id, r.refName)} className="border-vault-border text-jade-ink hover:bg-verify-green/10">
   310→                    <BellRing className="mr-1.5 h-3.5 w-3.5" /> Nudge
   311→                  </Button>
   312→                  <SwapDialog onSwap={(repl) => swap(r.id, repl)} />
   313→                </>
   314→              )}
   315→              {!isDone && (
   316→                <Button size="sm" variant="ghost" className="text-jade-muted hover:text-jade-ink"
   317→                  onClick={() => navigator.clipboard?.writeText(r.refLink).then(() => toast({ title: "Link copied" }))}>
   318→                  <Copy className="mr-1.5 h-3.5 w-3.5" /> Copy link
   319→                </Button>
   320→              )}
   321→              {!isDone && (
   322→                <Button size="sm" variant="ghost" className="text-verify-ink hover:text-verify-deep"
   323→                  onClick={() => window.open(`/?r=${tokenFromLink(r.refLink)}`, "_blank")}>
   324→                  <ExternalLink className="mr-1.5 h-3.5 w-3.5" /> Open form
   325→                </Button>
   326→              )}
   327→              {!isDone && (
   328→                <span className="ml-auto rounded-md bg-[#f0f6f2] px-2.5 py-1 font-mono text-xs text-[#33565b]" title="Reference uses this over the phone for identity verification">
   329→                  Callback code: {r.callbackCode}
   330→                </span>
   331→              )}
   332→            </div>
   333→            {!isDone && r.reminderCount > 0 && (
   334→              <p className="mt-2 text-xs text-[#8aa29c]">{r.reminderCount} automated reminder{r.reminderCount > 1 ? "s" : ""} sent · link expires {fmt(r.expiresAt)}</p>
   335→            )}
   336→          </motion.div>
   337→        );
   338→      })}
   339→
   340→      <p className="text-center text-[11px] text-[#8aa29c]">
   341→        References are matched to this vault by your account email ({email}).
   342→      </p>
   343→    </div>
   344→  );

   347→// ── Share a completed reference: one-time or duration link, candidate-controlled ──
   348→function RefShareDialog({ request, onCreated }: { request: { id: string; refName: string }; onCreated: () => void }) {
   349→  const { toast } = useToast();
   350→  const [open, setOpen] = useState(false);
   351→  const [busy, setBusy] = useState(false);
   352→  const [accessType, setAccessType] = useState<"ONE_TIME" | "DURATION">("DURATION");
   353→  const [days, setDays] = useState(14);
   354→  const [label, setLabel] = useState("");
   355→  const [created, setCreated] = useState<{ token: string; expiresAt: string | null } | null>(null);
   356→
   357→  const create = async () => {
   358→    setBusy(true);
   359→    try {
   360→      const res = await fetch("/api/reference/share", {
   361→        method: "POST", headers: { "Content-Type": "application/json" },
   362→        body: JSON.stringify({ action: "create", requestId: request.id, accessType, durationDays: accessType === "DURATION" ? days : undefined, label }),
   363→      });
   364→      const d = await res.json();
   365→      if (!res.ok || !d.ok) throw new Error(d.error ?? "Could not create the link");
   366→      setCreated({ token: d.link.token, expiresAt: d.link.expiresAt });
   367→      toast({ title: `Share link ready for ${request.refName}'s report` });
   368→      onCreated();
   369→    } catch (e) {
   370→      toast({ title: e instanceof Error ? e.message : "Could not create the link", variant: "destructive" });
   371→    } finally {
   372→      setBusy(false);
   373→    }
   374→  };
   375→
   376→  const link = created ? `${typeof window !== "undefined" ? window.location.origin : ""}/?rs=${created.token}` : "";
   377→
   378→  return (
   379→    <Dialog open={open} onOpenChange={(v) => { setOpen(v); if (!v) setCreated(null); }}>
   380→      <DialogTrigger asChild>
   381→        <Button size="sm" className="bg-verify-green text-vault-dark hover:bg-verify-green/90">
   382→          <Share2 className="mr-1.5 h-3.5 w-3.5" /> Share
   383→        </Button>
   384→      </DialogTrigger>
   385→      <DialogContent className="sm:max-w-md">
   386→        <DialogHeader>
   387→          <DialogTitle>Share {request.refName}&apos;s verified reference</DialogTitle>
   388→        </DialogHeader>
   389→        {!created ? (
   390→          <>
   391→            <p className="text-sm text-jade-muted">Anyone with the link sees the signed report — you keep control and can revoke it anytime from Controlled sharing.</p>
   392→            <div className="mt-1 space-y-2">
   393→              <p className="text-xs font-semibold uppercase tracking-wider text-jade-muted">Access rule</p>
   394→              <div className="grid grid-cols-2 gap-2">
   395→                <button type="button" onClick={() => setAccessType("DURATION")}
   396→                  className={cn("rounded-xl border p-3 text-left text-xs transition", accessType === "DURATION" ? "border-verify-green bg-verify-green/10 text-verify-ink" : "border-vault-border text-jade-muted hover:border-verify-green/40")}>
   397→                  <p className="text-sm font-semibold">Open for a period</p>
   398→                  <p className="mt-0.5">Stays open until the expiry you pick</p>
   399→                </button>
   400→                <button type="button" onClick={() => setAccessType("ONE_TIME")}
   401→                  className={cn("rounded-xl border p-3 text-left text-xs transition", accessType === "ONE_TIME" ? "border-verify-green bg-verify-green/10 text-verify-ink" : "border-vault-border text-jade-muted hover:border-verify-green/40")}>
   402→                  <p className="text-sm font-semibold">One-time view</p>
   403→                  <p className="mt-0.5">Closes after the first opening</p>
   404→                </button>
   405→              </div>
   406→              {accessType === "DURATION" && (
   407→                <div className="flex flex-wrap items-center gap-2">
   408→                  {[1, 7, 14, 30].map((d) => (
   409→                    <button key={d} type="button" onClick={() => setDays(d)}
