     1→     1→/**
     2→     2→ * VaultVerify - Skills Checklist PDF (Node / PDFKit port)
     3→     3→ *
     4→     4→ * Faithful port of the approved reference implementation (generate_checklist.py).
     5→     5→ * All coordinates are PDF points on US Letter (612 x 792) with a TOP-LEFT origin,
     6→     6→ * which is what PDFKit uses, so the `top` values from the reference apply unchanged.
     7→     7→ *
     8→     8→ * Usage:
     9→     9→ *   import { renderSkillsChecklistPdf } from "@/lib/pdf/skills-checklist-pdf";
    10→    10→ *   const pdf: Buffer = await renderSkillsChecklistPdf(data);
    11→    11→ *
    12→    12→ * Runtime: Node.js only (never the Edge runtime).
    13→    13→ *
    14→    14→ * Two documented additive extensions over the approved handoff (everything else is
    15→    15→ * unchanged; the golden test against reference/approved-design.pdf still passes):
    16→    16→ *  1. `rating: null` rows (candidate marked the skill N/A) render a dashed neutral
    17→    17→ *     ring with "N/A" centred - the same visual language as the design's own dashed
    18→    18→ *     N/A last-performed chip (#B9C3CD border, #8A97A3 text). Averages/proficient
    19→    19→ *     counts use rated rows only, exactly like the app's summarizeAnswers().
    20→    20→ *  2. `attestation.signatureImage` (data URL) embeds a Draw/Upload signature image
    21→    21→ *     above the signature line; typed signatures keep the Lora Italic text.
    22→    22→ */
    23→    23→import path from "node:path";

   108→   108→// --------------------------------------------------------------------------- colour helpers
   109→   109→type RGB = [number, number, number];
   110→   110→const toRgb = (hex: string): RGB => {
   111→   111→  const h = hex.replace("#", "");
   112→   112→  return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)];
   113→   113→};
   114→   114→/** Blend a colour toward white by k (0..1). */
   115→   115→const mix = (hex: string, k: number): RGB => toRgb(hex).map((v) => v + (255 - v) * k) as RGB;
   116→   116→const darken = (hex: string, k: number): RGB => toRgb(hex).map((v) => v * k) as RGB;

   118→   118→// --------------------------------------------------------------------------- validation
   119→   119→function validate(data: ChecklistData) {
   120→   120→  for (const cat of data.categories) {
   121→   121→    for (const s of cat.skills) {
   122→   122→      if (s.rating != null && ![1, 2, 3, 4].includes(s.rating)) throw new Error(`Bad rating for "${s.name}": ${s.rating}`);
   123→   123→      if (!(s.lastPerformed in REC)) throw new Error(`Bad lastPerformed for "${s.name}": ${s.lastPerformed}`);
   124→   124→    }
   125→   125→  }
   126→   126→  if (!data.categories.some((c) => c.skills.length)) throw new Error("Checklist has no skills");
   127→   127→}
   128→   128→
   129→   129→// --------------------------------------------------------------------------- public API
   130→   130→export async function renderSkillsChecklistPdf(data: ChecklistData, opts: RenderOptions = {}): Promise<Buffer> {
   131→   131→  validate(data);
   132→   132→  const assetsDir = opts.assetsDir ?? path.join(process.cwd(), "assets", "skills-checklist");
   133→   133→  // Two passes: the first counts pages so "Page x of y" is right on every page.
   134→   134→  const first = await draw(data, assetsDir, 1);
   135→   135→  const second = await draw(data, assetsDir, first.pages);
   136→   136→  return second.buffer;
   137→   137→}

   139→   139→// --------------------------------------------------------------------------- renderer
   140→   140→function draw(data: ChecklistData, assetsDir: string, total: number): Promise<{ buffer: Buffer; pages: number }> {
   141→   141→  const cand = data.candidate;
   142→   142→  const cats = data.categories;
   143→   143→  const all = cats.flatMap((c) => c.skills);
   144→   144→  const N = all.length;
   145→   145→  const ratedAll = all.filter((s) => s.rating != null);
   146→   146→  const AVG = ratedAll.length ? ratedAll.reduce((a, s) => a + (s.rating as Rating), 0) / ratedAll.length : 0;
   147→   147→  const PROF = all.filter((s) => s.rating === 4).length;
   148→   148→  const RECENT = all.filter((s) => s.lastPerformed === "within_3_months").length;
   149→   149→  const cnt = { 1: 0, 2: 0, 3: 0, 4: 0 } as Record<Rating, number>;
   150→   150→  all.forEach((s) => { if (s.rating != null) cnt[s.rating]++; });
   151→   151→  const rcnt = { within_3_months: 0, within_6_months: 0, over_6_months: 0, na: 0 } as Record<LastPerformed, number>;
   152→   152→  all.forEach((s) => rcnt[s.lastPerformed]++);
   153→   153→  const roleLine = `${cand.name}, ${cand.jobTitle} (${cand.specialty})`;
   154→   154→
   155→   155→  const doc = new PDFDocument({
   156→   156→    size: [W, H],
   157→   157→    margin: 0,
   158→   158→    autoFirstPage: false,
   159→   159→    info: { Title: `Skills Checklist - ${cand.name} (${cand.jobTitle}, ${cand.specialty})`, Author: "VaultVerify" },
   160→   160→  });
   161→   161→  const chunks: Buffer[] = [];
   162→   162→  doc.on("data", (c: Buffer) => chunks.push(c));
   163→   163→
   164→   164→  const fontDir = path.join(assetsDir, "fonts");
   165→   165→  doc.registerFont("P", path.join(fontDir, "Poppins-Regular.ttf"));
   166→   166→  doc.registerFont("PL", path.join(fontDir, "Poppins-Light.ttf"));
   167→   167→  doc.registerFont("PM", path.join(fontDir, "Poppins-Medium.ttf"));
   168→   168→  doc.registerFont("PB", path.join(fontDir, "Poppins-Bold.ttf"));
   169→   169→  doc.registerFont("Sig", path.join(fontDir, "Lora-Italic-Regular.ttf"));
   170→   170→  // Pass the file PATH (not a Buffer) so PDFKit embeds the logo once and reuses it on every page
   171→   171→  const logo = path.join(assetsDir, "logo.png");
   172→   172→
   173→   173→  // ------------------------------------------------------------------ primitives
   174→   174→  type Fill = string | RGB;
   175→   175→  const rrect = (x: number, top: number, w: number, h: number, r: number, fill?: Fill, stroke?: Fill, lw = 0.8) => {
   176→   176→    doc.roundedRect(x, top, w, h, r);
   177→   177→    if (fill && stroke) { doc.lineWidth(lw); doc.fillAndStroke(fill as any, stroke as any); }
   178→   178→    else if (fill) doc.fill(fill as any);
   179→   179→    else if (stroke) { doc.lineWidth(lw); doc.stroke(stroke as any); }
   180→   180→  };
   181→   181→  const width = (s: string, font = "P", size = 9) => doc.font(font).fontSize(size).widthOfString(s);
   182→   182→  const text = (x: number, top: number, s: string, font = "P", size = 9, color: Fill = INK, align: "l" | "r" | "c" = "l") => {
   183→   183→    doc.font(font).fontSize(size).fillColor(color as any);
   184→   184→    const w = doc.widthOfString(s);
   185→   185→    const x0 = align === "l" ? x : align === "r" ? x - w : x - w / 2;
   186→   186→    doc.text(s, x0, top, { lineBreak: false, baseline: "alphabetic" });
   187→   187→  };
   188→   188→  const line = (x1: number, y1: number, x2: number, y2: number, color: Fill, lw: number) => {
   189→   189→    doc.lineWidth(lw).strokeColor(color as any).moveTo(x1, y1).lineTo(x2, y2).stroke();
   190→   190→  };
   191→   191→  /** ReportLab simpleSplit: greedy word wrap on spaces. */
   192→   192→  const wrap = (s: string, font: string, size: number, maxW: number): string[] => {
   193→   193→    const out: string[] = [];
   194→   194→    for (const para of s.split("\n")) {
   195→   195→      let cur = "";
   196→   196→      for (const w of para.split(/\s+/).filter(Boolean)) {
   197→   197→        const cand2 = cur ? `${cur} ${w}` : w;
   198→   198→        if (!cur || width(cand2, font, size) <= maxW) cur = cand2;
   199→   199→        else { out.push(cur); cur = w; }
   200→   200→      }
   201→   201→      out.push(cur);
   202→   202→    }
   203→   203→    return out;
   204→   204→  };
   205→   205→  const alpha = (fn: () => void, a: number) => { doc.save(); doc.fillOpacity(a); fn(); doc.restore(); };
   206→   206→
   207→   207→  const headerRings = (hb: number) => {
   208→   208→    doc.save();
   209→   209→    doc.rect(0, 0, W, hb).clip();
   210→   210→    for (const [r, a] of [[150, 0.06], [112, 0.07], [74, 0.08]] as const) {
   211→   211→      doc.fillColor([77, 171, 87] as any).fillOpacity(a).circle(W - 40, 10, r).fill();
   212→   212→    }
   213→   213→    doc.restore();
   214→   214→  };
   215→   215→
   216→   216→  // ------------------------------------------------------------------ rating widgets
   217→   217→  /** 4-segment score ring; segments fill clockwise from 12 o'clock up to `level`, number centred. */
   218→   218→  const ring = (cx: number, cy: number, level: Rating, r = 8.2, lw = 2.6) => {
   219→   219→    doc.lineCap("round").lineWidth(lw);
   220→   220→    for (let i = 0; i < 4; i++) {
   221→   221→      const a0 = ((i * 90 + 6) * Math.PI) / 180;
   222→   222→      const a1 = ((i * 90 + 84) * Math.PI) / 180;
   223→   223→      const p = (a: number): [number, number] => [cx + r * Math.sin(a), cy - r * Math.cos(a)];
   224→   224→      const [x0, y0] = p(a0);
   225→   225→      const [x1, y1] = p(a1);
   226→   226→      doc.path(`M ${x0} ${y0} A ${r} ${r} 0 0 1 ${x1} ${y1}`).stroke((i < level ? LVL[level] : "#E3EAE7") as any);
   227→   227→    }
   228→   228→    text(cx, cy + 2.6, String(level), "PB", 7.4, LVL[level], "c");
   229→   229→  };
   230→   230→  /** N/A rating row (additive): dashed neutral ring, same language as the design's dashed N/A chip. */
   231→   231→  const ringNA = (cx: number, cy: number, r = 8.2) => {
   232→   232→    doc.lineWidth(1.8).dash(2.2, { space: 1.8 });
   233→   233→    doc.circle(cx, cy, r).stroke("#B9C3CD");
   234→   234→    doc.undash();
   235→   235→    text(cx, cy + 2.2, "N/A", "PB", 5.4, "#8A97A3", "c");
   236→   236→  };
   237→   237→  const levelChip = (x: number, cy: number, level: Rating) => {
   238→   238→    const lab = LVL_LABEL[level];
   239→   239→    const w = width(lab, "PB", 6.6) + 14;
   240→   240→    rrect(x, cy - 6.5, w, 13, 6.5, mix(LVL[level], 0.84));
   241→   241→    text(x + w / 2, cy + 2.4, lab, "PB", 6.6, darken(LVL[level], 0.72), "c");
   242→   242→  };
   243→   243→  const recChip = (x: number, cy: number, key: LastPerformed, size = 6.4): number => {
   244→   244→    const st = REC[key];
   245→   245→    const icon = key === "na" ? 0 : 10;
   246→   246→    const w = width(st.label, "PB", size) + 14 + icon;
   247→   247→    rrect(x, cy - 6.5, w, 13, 6.5, st.fill);
   248→   248→    if (st.border) {
   249→   249→      doc.lineWidth(0.8).dash(2, { space: 1.6 });
   250→   250→      doc.roundedRect(x, cy - 6.5, w, 13, 6.5).stroke(st.border);
   251→   251→      doc.undash();
   252→   252→    }
   253→   253→    if (icon) { // tiny clock icon
   254→   254→      const ix = x + 9;
   255→   255→      doc.lineWidth(0.8).lineCap("round").strokeColor(st.text);
   256→   256→      doc.circle(ix, cy, 2.8).stroke();
   257→   257→      doc.moveTo(ix, cy).lineTo(ix, cy - 1.8).stroke();
   258→   258→      doc.moveTo(ix, cy).lineTo(ix + 1.5, cy + 0.6).stroke();
   259→   259→    }
   260→   260→    text(icon ? x + (w + icon) / 2 : x + w / 2, cy + 2.3, st.label, "PB", size, st.text, "c");
   261→   261→    return w;
   262→   262→  };
   263→   263→
   264→   264→  // ------------------------------------------------------------------ page chrome
   265→   265→  let page = 0;
   266→   266→  const tableHeader = (top: number): number => {
   267→   267→    rrect(M, top, CW, 34, 10, INK);
   268→   268→    text(M + 16, top + 21, "Skill", "PB", 8.5, WHITE);
   269→   269→    text(RX - 10, top + 21, "Rating", "PB", 8.5, ACCENT_ON_DARK);
   270→   270→    text(LPX, top + 21, "Last performed", "PB", 8.5, ACCENT_ON_DARK);
   271→   271→    return top + 40;
   272→   272→  };
   273→   273→  const footer = (last: boolean) => {
   274→   274→    line(M, H - 56, W - M, H - 56, LINE, 0.6);
   275→   275→    if (last) {
   276→   276→      wrap(data.footerNote, "P", 6.6, CW - 70).forEach((ln, k) => text(M, H - 44 + k * 9, ln, "P", 6.6, SLATE));
   277→   277→      let gen = `Generated ${data.generatedOn} by VaultVerify.`;
   278→   278→      if (data.sampleNotice) gen += "   " + data.sampleNotice;
   279→   279→      text(M, H - 44 + 2 * 9, gen, "PM", 6.6, TEAL);
   280→   280→    } else {
   281→   281→      text(M, H - 40, `VaultVerify   Skills Checklist: ${roleLine}`, "PM", 6.8, TEAL);
   282→   282→    }
   283→   283→    text(W - M, H - 40, `Page ${page} of ${total}`, "PM", 6.8, SLATE, "r");
   284→   284→  };
   285→   285→  const slimHeader = (): number => {
   286→   286→    doc.rect(0, 0, W, 52).fill(DEEP);
   287→   287→    headerRings(52);
   288→   288→    doc.rect(0, 52, W, 2).fill(LEAF);
   289→   289→    rrect(M, 11, 30, 30, 8, "#F3F8F4");
   290→   290→    doc.image(logo, M + 3, 14, { width: 24, height: 24 });
   291→   291→    text(M + 40, 31, "VaultVerify", "PB", 12.5, WHITE);
   292→   292→    text(W - M, 31, `Skills Checklist: ${roleLine}`, "PM", 8.5, WHITE, "r");
   293→   293→    return 72;
   294→   294→  };
   295→   295→  const newPage = (): number => {
   296→   296→    footer(false);
   297→   297→    doc.addPage();
   298→   298→    page++;
   299→   299→    return slimHeader();
   300→   300→  };
   301→   301→
   302→   302→  // ================================================================== PAGE 1
   303→   303→  doc.addPage();
   304→   304→  page = 1;
   305→   305→
   306→   306→  // ---- brand header (logo LEFT, platform name next to it)
   307→   307→  const HB = 104;
   308→   308→  doc.rect(0, 0, W, HB).fill(DEEP);
   309→   309→  headerRings(HB);
   310→   310→  doc.rect(0, HB, W, 3).fill(LEAF);
   311→   311→  rrect(M, 17, 54, 54, 14, "#F3F8F4");
   312→   312→  doc.image(logo, M + 4, 21, { width: 46, height: 46 });
   313→   313→  text(M + 68, 43, "VaultVerify", "PB", 21, WHITE);
   314→   314→  text(M + 69, 60, "References, verified. Skills, proven.", "PL", 8.5, "#BFE3C7");
   315→   315→  text(W - M, 40, "Skills Checklist", "PM", 15, WHITE, "r");
   316→   316→  text(W - M, 56, "Self-assessment completed by the candidate", "PL", 8, "#BFD6D3", "r");
   317→   317→
   318→   318→  // ---- candidate card (overlaps the header)
   319→   319→  const CT = 78;
   320→   320→  const CH = 92;
   321→   321→  [0.025, 0.03, 0.04].forEach((a, i) => { // faux soft shadow
   322→   322→    alpha(() => doc.roundedRect(M - 3 + i, CT + 1 + (2 - i) * 1.2, CW + 6 - 2 * i, CH + 2, 16).fill([10, 46, 51] as any), a);
   323→   323→  });
   324→   324→  rrect(M, CT, CW, CH, 14, WHITE, LINE);
   325→   325→  doc.circle(M + 36, CT + 36, 20).fill(TEAL);
   326→   326→  text(M + 36, CT + 43, cand.name.slice(0, 1).toUpperCase(), "PB", 18, WHITE, "c");
   327→   327→  // Name: shrink to fit left of the divider (max 160pt); ellipsis only as a last resort
   328→   328→  let nameTxt = cand.name;
   329→   329→  let nameSize = 21;
   330→   330→  while (width(nameTxt, "PB", nameSize) > 160 && nameSize > 13) nameSize--;
   331→   331→  while (width(nameTxt, "PB", nameSize) > 160 && nameTxt.length > 4) nameTxt = nameTxt.slice(0, -4).trimEnd() + "...";
   332→   332→  text(M + 68, CT + 34, nameTxt, "PB", nameSize, INK);
   333→   333→  let cx = M + 68;
   334→   334→  for (const chip of [cand.jobTitle, cand.specialty, cand.profession]) {
   335→   335→    const w = width(chip, "PM", 7.5) + 14;
   336→   336→    if (cx + w > M + 232) break; // never run into the divider
   337→   337→    rrect(cx, CT + 44, w, 16, 8, MINT2);
   338→   338→    text(cx + w / 2, CT + 55.2, chip, "PM", 7.5, TEAL, "c");
   339→   339→    cx += w + 5;
   340→   340→  }
   341→   341→  line(M + 240, CT + 16, M + 240, CT + CH - 16, LINE, 0.8);
   342→   342→  const meta = (x: number, top: number, label: string, value: string) => {
   343→   343→    text(x, top, label, "P", 6.8, SLATE);
   344→   344→    text(x, top + 12, value, "PM", 8, INK);
   345→   345→  };
   346→   346→  const mx = M + 256;
   347→   347→  meta(mx, CT + 30, "Email", cand.email);
   348→   348→  meta(mx + 150, CT + 30, "Self-reported experience", cand.experience);
   349→   349→  meta(mx, CT + 62, "Completed", cand.completedOn);
   350→   350→  meta(mx + 68, CT + 62, "Valid until", cand.validUntil);
   351→   351→  meta(mx + 160, CT + 62, "Source", cand.source);
   352→   352→

   390→   390→  const bw = (CW - 40 - 28) / 2;
   391→   391→  text(M + 20, ST0 + 98, "Rating mix", "PM", 7, SLATE);
   392→   392→  stacked(M + 20, ST0 + 103, bw, 10, [[cnt[4], LVL[4], WHITE], [cnt[3], LVL[3], WHITE], [cnt[2], LVL[2], WHITE], [cnt[1], LVL[1], WHITE]]);
   393→   393→  text(M + 20 + bw + 28, ST0 + 98, "Last performed", "PM", 7, SLATE);
   394→   394→  stacked(M + 20 + bw + 28, ST0 + 103, bw, 10, [
   395→   395→    [rcnt.within_3_months, REC.within_3_months.bar, WHITE],
   396→   396→    [rcnt.within_6_months, REC.within_6_months.bar, WHITE],
   397→   397→    [rcnt.over_6_months, REC.over_6_months.bar, "#3F4B5A"],

   407→   407→  let yy = OT + 44;
   408→   408→  for (const cat of cats) {
   409→   409→    const rs = (cat.skills.filter((s) => s.rating != null) as { rating: Rating }[]).map((s) => s.rating);

   411→   411→    const pr = rs.filter((r) => r === 4).length;
   412→   412→    const rc = cat.skills.filter((s) => s.lastPerformed === "within_3_months").length;
   413→   413→    let label = cat.shortName || cat.name;
   414→   414→    while (width(label, "PM", 7.6) > 150 && label.length > 4) label = label.slice(0, -4).trimEnd() + "...";
   415→   415→    text(M + 18, yy + 4, label, "PM", 7.6, INK);
   416→   416→    const tx = M + 176;
   417→   417→    const tw = 110;
   418→   418→    doc.roundedRect(tx, yy - 3, tw, 6, 3).fill("#C9DDD0");
   419→   419→    doc.roundedRect(tx, yy - 3, (tw * avg) / 4, 6, 3).fill(LEAF);

   422→   422→    text(W - M - 18, yy + 4, `${rc} of ${cat.skills.length} recent`, "P", 7.2, SLATE, "r");

   424→   424→  }

   431→   431→  ([1, 2, 3, 4] as Rating[]).forEach((lv) => {

   493→   493→
