    22→    22→function hexToRgb(hex: string): RGB {
    23→    23→  const h = hex.replace("#", "");
    24→    24→  const n = parseInt(h.length === 3 ? h.split("").map((c) => c + c).join("") : h, 16);
    25→    25→  return rgb(((n >> 16) & 255) / 255, ((n >> 8) & 255) / 255, (n & 255) / 255);
    26→    26→}

   117→   117→// SVG helpers (design-space units, drawn with scale S)
   118→   118→function pt(r: number, deg: number): [number, number] {
   119→   119→  const rad = ((deg - 90) * Math.PI) / 180;
   120→   120→  return [r * Math.cos(rad), r * Math.sin(rad)];
   121→   121→}
   122→   122→function wedgePath(ro: number, ri: number, d1: number, d2: number): string {
   123→   123→  const large = d2 - d1 > 180 ? 1 : 0;
   124→   124→  const f = (n: number) => n.toFixed(2);
   125→   125→  const [x1, y1] = pt(ro, d1), [x2, y2] = pt(ro, d2);
   126→   126→  const [x3, y3] = pt(ri, d2), [x4, y4] = pt(ri, d1);
   127→   127→  return `M ${f(x1)} ${f(y1)} A ${f(ro)} ${f(ro)} 0 ${large} 1 ${f(x2)} ${f(y2)} L ${f(x3)} ${f(y3)} A ${f(ri)} ${f(ri)} 0 ${large} 0 ${f(x4)} ${f(y4)} Z`;
   128→   128→}
   129→   129→function rrectPath(w: number, h: number, r: number): string {
   130→   130→  const rr = Math.min(r, w / 2, h / 2);
   131→   131→  const f = (n: number) => n.toFixed(2);
   132→   132→  return `M ${f(rr)} 0 H ${f(w - rr)} A ${f(rr)} ${f(rr)} 0 0 1 ${f(w - rr)} ${f(rr)} V ${f(h - rr)} A ${f(rr)} ${f(rr)} 0 0 1 ${f(w - rr)} ${f(h)} H ${f(rr)} A ${f(rr)} ${f(rr)} 0 0 1 0 ${f(h - rr)} V ${f(rr)} A ${f(rr)} ${f(rr)} 0 0 1 ${f(rr)} 0 Z`;
   133→   133→}

   135→   135→async function embedSignatureImage(pdf: PDFDocument, dataUrl: string): Promise<PDFImage | null> {
   136→   136→  try {
   137→   137→    const m = /^data:image\/(png|jpe?g);base64,([\s\S]+)$/.exec(dataUrl.trim());
   138→   138→    if (!m) return null;
   139→   139→    const buf = Buffer.from(m[2], "base64");
   140→   140→    return /^png$/i.test(m[1]) ? await pdf.embedPng(buf) : await pdf.embedJpg(buf);
   141→   141→  } catch { return null; }
   142→   142→}

   172→   172→    }
