    39→// ── Share dialog ──
    40→function ShareDialog({ completion, onClose, onCreated }: { completion: CompletionDto; onClose: () => void; onCreated: () => void }) {
    41→  const [preset, setPreset] = useState<string>("ONE_TIME");
    42→  const [customDays, setCustomDays] = useState("30");
    43→  const [label, setLabel] = useState("");
    44→  const [created, setCreated] = useState<{ token: string; accessType: string; durationDays: number | null; expiresAt: string | null } | null>(null);
    45→  const [busy, setBusy] = useState(false);
    46→  const [copied, setCopied] = useState(false);
    47→
    48→  const create = async () => {
    49→    setBusy(true);
    50→    try {
    51→      const chosen = SHARE_PRESETS.find((p) => p.key === preset)!;
    52→      const accessType = chosen.key === "ONE_TIME" ? "ONE_TIME" : "DURATION";
    53→      const durationDays = chosen.key === "ONE_TIME" ? null : chosen.key === "custom" ? Number(customDays) : chosen.days;
    54→      const res = await fetch("/api/checklist/share", {
    55→        method: "POST", headers: { "Content-Type": "application/json" },
    56→        body: JSON.stringify({ completionId: completion.id, accessType, durationDays, label }),
    57→      });
    58→      const d = await res.json();
    59→      if (d.ok) { setCreated(d.link); onCreated(); } else throw new Error(d.error);
    60→    } catch { /* inline: button re-enables */ } finally { setBusy(false); }
    61→  };
    62→
    63→  const link = created ? `${window.location.origin}/?s=${created.token}` : "";
    64→
    65→  return (
    66→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
    67→      className="fixed inset-0 z-50 flex items-center justify-center bg-[#0b2b30]/45 p-4 backdrop-blur-sm" onClick={onClose}>
    68→      <motion.div initial={{ scale: 0.96, y: 10 }} animate={{ scale: 1, y: 0 }} transition={{ type: "spring", stiffness: 300, damping: 24 }}
    69→        className="w-full max-w-md rounded-2xl border border-vault-border bg-white vv-card-shadow p-6" onClick={(e) => e.stopPropagation()}>
    70→        <div className="flex items-start justify-between">
    71→          <div>
    72→            <h3 className="flex items-center gap-2 text-base font-semibold text-jade-ink"><Link2 className="h-4 w-4 text-verify-ink" /> Share checklist</h3>
    73→            <p className="mt-0.5 text-xs text-jade-muted">{completion.specialtyLabel} · {completion.jobTitle}</p>
    74→          </div>
    75→          <button type="button" onClick={onClose} aria-label="Close" className="rounded-lg p-1.5 text-jade-muted hover:bg-jade-ink/5 hover:text-jade-ink"><X className="h-4 w-4" /></button>
    76→        </div>
    77→
    78→        {!created ? (
    79→          <>
    80→            <p className="mt-4 text-xs font-semibold uppercase tracking-wider text-jade-muted">Who can open it & for how long</p>
    81→            <div className="mt-2 grid grid-cols-2 gap-2">
    82→              {SHARE_PRESETS.map((p) => (
    83→                <button key={p.key} type="button" onClick={() => setPreset(p.key)}
    84→                  className={cn("rounded-xl border px-3 py-2.5 text-left text-sm transition",
    85→                    preset === p.key ? "border-verify-green bg-verify-green/15 text-verify-ink" : "border-vault-border text-jade-muted hover:text-jade-ink")}>
    86→                  <span className="block font-semibold">{p.label}</span>
    87→                  <span className="mt-0.5 block text-[10px] opacity-75">
    88→                    {p.key === "ONE_TIME" ? "Viewed once, then dead" : p.key === "custom" ? "You pick the days" : p.days === 1 ? "24 hours" : `${p.days} days from now`}
    89→                  </span>
    90→                </button>
    91→              ))}
    92→            </div>
    93→            {preset === "custom" && (
    94→              <div className="mt-3">
    95→                <Label className="text-jade-ink/80">Days the link stays open</Label>
    96→                <Input type="number" min={1} max={365} value={customDays} onChange={(e) => setCustomDays(e.target.value)}
    97→                  className="mt-1 border-vault-border bg-white text-jade-ink" />
    98→              </div>
    99→            )}
   100→            <div className="mt-3">
   101→              <Label className="text-jade-ink/80">Note for the recipient (optional)</Label>
   102→              <Input value={label} onChange={(e) => setLabel(e.target.value)} placeholder="e.g. For the SNF recruiter"
   103→                className="mt-1 border-vault-border bg-white text-jade-ink placeholder:text-[#8aa29c]" />
   104→            </div>
   105→            <Button onClick={create} disabled={busy} className="mt-5 w-full bg-verify-green text-vault-dark hover:bg-verify-green/90">
   106→              {busy ? "Creating…" : "Create share link"}
   107→            </Button>
   108→          </>
   109→        ) : (
   110→          <>
   111→            <div className="mt-4 rounded-xl border border-verify-green/30 bg-verify-green/10 p-4">
   112→              <p className="flex items-center gap-2 text-sm font-semibold text-verify-ink"><BadgeCheck className="h-4 w-4" /> Link ready</p>
   113→              <p className="mt-1 text-xs text-[#33565b]">
   114→                Access: <span className="font-semibold">{shareAccessLabel(created)}</span>
   115→                {created.expiresAt && created.accessType === "DURATION" ? ` — expires ${fmt(created.expiresAt)}` : created.accessType === "ONE_TIME" ? " — first view burns the link" : ""}
   116→              </p>
   117→            </div>
   118→            <div className="mt-3 break-all rounded-xl border border-vault-border bg-[#eef4f1] p-3 font-mono text-xs text-jade-ink">{link}</div>
   119→            <div className="mt-4 flex gap-2">
   120→              <Button onClick={() => { navigator.clipboard.writeText(link).then(() => { setCopied(true); setTimeout(() => setCopied(false), 1600); }); }}
   121→                className="flex-1 bg-verify-green text-vault-dark hover:bg-verify-green/90">
   122→                {copied ? <><Check className="mr-1.5 h-4 w-4" /> Copied</> : <><Copy className="mr-1.5 h-4 w-4" /> Copy link</>}
   123→              </Button>
   124→              <Button variant="ghost" onClick={onClose} className="border border-vault-border text-jade-muted hover:text-jade-ink">Done</Button>
   125→            </div>
   126→            <p className="mt-3 text-[11px] leading-relaxed text-[#8aa29c]">
   127→              Send this to any recruiter. They’ll see the access type and deadline on the page, and can download the PDF.
   128→            </p>
   129→          </>
   130→        )}
   131→      </motion.div>
   132→    </motion.div>
   133→  );
   134→}

   695→// ── Catalog picker (candidate requests a checklist → superadmin approves) ──
   696→function RequestPanel({ sets, onClose, onRequested }: { sets: CatalogSet[]; onClose: () => void; onRequested: () => void }) {
   697→  const [busyId, setBusyId] = useState<string | null>(null);
   698→  const [error, setError] = useState<string | null>(null);
   699→
   700→  const request = async (s: CatalogSet) => {
   701→    setBusyId(s.key);
   702→    setError(null);
   703→    try {
   704→      const res = await fetch("/api/checklist/request", {
   705→        method: "POST", headers: { "Content-Type": "application/json" },
   706→        body: JSON.stringify({ profession: s.profession, jobTitle: s.jobTitle, specialty: s.specialty }),
   707→      });
   708→      const d = await res.json();
   709→      if (!res.ok || !d.ok) throw new Error(d.error ?? "Request failed");
   710→      onRequested();
   711→    } catch (e) {
   712→      setError(e instanceof Error ? e.message : "Request failed");
   713→    } finally { setBusyId(null); }
