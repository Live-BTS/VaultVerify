    29→// ── Share dialog ──
    30→function ShareDialog({ completion, onClose, onCreated }: { completion: CompletionDto; onClose: () => void; onCreated: () => void }) {
    31→  const [preset, setPreset] = useState<string>("ONE_TIME");
    32→  const [customDays, setCustomDays] = useState("30");
    33→  const [label, setLabel] = useState("");
    34→  const [created, setCreated] = useState<{ token: string; accessType: string; durationDays: number | null; expiresAt: string | null } | null>(null);
    35→  const [busy, setBusy] = useState(false);
    36→  const [copied, setCopied] = useState(false);
    37→
    38→  const create = async () => {
    39→    setBusy(true);
    40→    try {
    41→      const chosen = SHARE_PRESETS.find((p) => p.key === preset)!;
    42→      const accessType = chosen.key === "ONE_TIME" ? "ONE_TIME" : "DURATION";
    43→      const durationDays = chosen.key === "ONE_TIME" ? null : chosen.key === "custom" ? Number(customDays) : chosen.days;
    44→      const res = await fetch("/api/checklist/share", {
    45→        method: "POST", headers: { "Content-Type": "application/json" },
    46→        body: JSON.stringify({ completionId: completion.id, accessType, durationDays, label }),
    47→      });
    48→      const d = await res.json();
    49→      if (d.ok) { setCreated(d.link); onCreated(); } else throw new Error(d.error);
    50→    } catch { /* toast-less: show inline error state via button re-enable */ } finally { setBusy(false); }
    51→  };
    52→
    53→  const link = created ? `${window.location.origin}/?s=${created.token}` : "";
    54→
    55→  return (
    56→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
    57→      className="fixed inset-0 z-50 flex items-center justify-center bg-vault-dark/80 p-4 backdrop-blur-sm" onClick={onClose}>
    58→      <motion.div initial={{ scale: 0.96, y: 10 }} animate={{ scale: 1, y: 0 }} transition={{ type: "spring", stiffness: 300, damping: 24 }}
    59→        className="w-full max-w-md rounded-2xl border border-vault-border bg-[#062024] p-6" onClick={(e) => e.stopPropagation()}>
    60→        <div className="flex items-start justify-between">
    61→          <div>
    62→            <h3 className="flex items-center gap-2 text-base font-semibold text-verify-light"><Link2 className="h-4 w-4 text-verify-green" /> Share checklist</h3>
    63→            <p className="mt-0.5 text-xs text-[#8fb0ab]">{completion.specialtyLabel} · {completion.jobTitle}</p>
    64→          </div>
    65→          <button type="button" onClick={onClose} aria-label="Close" className="rounded-lg p-1.5 text-[#8fb0ab] hover:bg-white/5 hover:text-verify-light"><X className="h-4 w-4" /></button>
    66→        </div>
    67→
    68→        {!created ? (
    69→          <>
    70→            <p className="mt-4 text-xs font-semibold uppercase tracking-wider text-[#8fb0ab]">Who can open it & for how long</p>
    71→            <div className="mt-2 grid grid-cols-2 gap-2">
    72→              {SHARE_PRESETS.map((p) => (
    73→                <button key={p.key} type="button" onClick={() => setPreset(p.key)}
    74→                  className={cn("rounded-xl border px-3 py-2.5 text-left text-sm transition",
    75→                    preset === p.key ? "border-verify-green bg-verify-green/15 text-verify-green" : "border-vault-border text-[#8fb0ab] hover:text-verify-light")}>
    76→                  <span className="block font-semibold">{p.label}</span>
    77→                  <span className="mt-0.5 block text-[10px] opacity-75">
    78→                    {p.key === "ONE_TIME" ? "Viewed once, then dead" : p.key === "custom" ? "You pick the days" : p.days === 1 ? "24 hours" : `${p.days} days from now`}
    79→                  </span>
    80→                </button>
    81→              ))}
    82→            </div>
    83→            {preset === "custom" && (
    84→              <div className="mt-3">
    85→                <Label className="text-verify-light/80">Days the link stays open</Label>
    86→                <Input type="number" min={1} max={365} value={customDays} onChange={(e) => setCustomDays(e.target.value)}
    87→                  className="mt-1 border-vault-border bg-vault-dark/60 text-verify-light" />
    88→              </div>
    89→            )}
    90→            <div className="mt-3">
    91→              <Label className="text-verify-light/80">Note for the recipient (optional)</Label>
    92→              <Input value={label} onChange={(e) => setLabel(e.target.value)} placeholder="e.g. For the SNF recruiter"
    93→                className="mt-1 border-vault-border bg-vault-dark/60 text-verify-light placeholder:text-[#5c7a76]" />
    94→            </div>
    95→            <Button onClick={create} disabled={busy} className="mt-5 w-full bg-verify-green text-vault-dark hover:bg-verify-green/90">
    96→              {busy ? "Creating…" : "Create share link"}
    97→            </Button>
    98→          </>
    99→        ) : (
   100→          <>
   101→            <div className="mt-4 rounded-xl border border-verify-green/30 bg-verify-green/10 p-4">
   102→              <p className="flex items-center gap-2 text-sm font-semibold text-verify-green"><BadgeCheck className="h-4 w-4" /> Link ready</p>
   103→              <p className="mt-1 text-xs text-[#c9ded9]">
   104→                Access: <span className="font-semibold">{shareAccessLabel(created)}</span>
   105→                {created.expiresAt && created.accessType === "DURATION" ? ` — expires ${fmt(created.expiresAt)}` : created.accessType === "ONE_TIME" ? " — first view burns the link" : ""}
   106→              </p>
   107→            </div>
   108→            <div className="mt-3 break-all rounded-xl border border-vault-border bg-vault-dark/70 p-3 font-mono text-xs text-verify-light">{link}</div>
   109→            <div className="mt-4 flex gap-2">
   110→              <Button onClick={() => { navigator.clipboard.writeText(link).then(() => { setCopied(true); setTimeout(() => setCopied(false), 1600); }); }}
   111→                className="flex-1 bg-verify-green text-vault-dark hover:bg-verify-green/90">
   112→                {copied ? <><Check className="mr-1.5 h-4 w-4" /> Copied</> : <><Copy className="mr-1.5 h-4 w-4" /> Copy link</>}
   113→              </Button>
   114→              <Button variant="ghost" onClick={onClose} className="border border-vault-border text-[#8fb0ab] hover:text-verify-light">Done</Button>
   115→            </div>
   116→            <p className="mt-3 text-[11px] leading-relaxed text-[#5c7a76]">
   117→              Send this to any recruiter. They’ll see the access type and deadline on the page, and can download the PDF.
   118→            </p>
   119→          </>
   120→        )}
   121→      </motion.div>
   122→    </motion.div>
   123→  );
   124→}

   142→
   143→  const loadMe = useCallback(async () => {
   144→    const res = await fetch("/api/checklist/account", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "me" }) });
   145→    const d: MeData & { ok: boolean } = await res.json();

   147→    return d.account;

   151→    (async () => {
   152→      try {
   153→        const res = await fetch("/api/checklists");

   156→        await loadMe();
   157→      } catch { /* portal still renders */ } finally { setBooting(false); }
   158→    })();

   161→  // claim an invite link after sign-in
   162→  useEffect(() => {
   163→    if (!me?.account || !inviteToken || claimedInvite === inviteToken) return;

   165→    fetch("/api/checklist/invite", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "claim", token: inviteToken }) })
   166→      .then(() => loadMe())
   167→      .catch(() => undefined);

   171→    e?.preventDefault();

   173→    setAuthError(null);

   175→      const res = await fetch("/api/checklist/account", {
   176→        method: "POST", headers: { "Content-Type": "application/json" },
   177→        body: JSON.stringify({ action: mode, ...form }),
   178→      });
   179→      const d = await res.json();
   180→      if (!res.ok || !d.ok) throw new Error(d.error ?? "Something went wrong");
   181→      await loadMe();
   182→      if (window.history?.replaceState) window.history.replaceState({}, "", "/");

   184→      setAuthError(err instanceof Error ? err.message : "Something went wrong");

   186→  };

   188→  const signOut = async () => {
   189→    await fetch("/api/checklist/account", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "logout" }) });

   192→

   196→      <ChecklistFill
   197→        spec={fillSpec}
   198→        sets={sets}
   199→        onExit={async () => { setFillSpec(null); await loadMe(); }}

   201→      />
   202→    );
   203→  }
   204→
   205→  // ── Auth gate ──
   206→  if (booting) {
   207→    return <div className="flex min-h-screen items-center justify-center bg-vault-dark"><Spinner label="Opening your vault…" /></div>;

   212→        <motion.div initial={{ opacity: 0, y: 14 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.45, ease: [0.22, 1, 0.36, 1] }}
   213→          className="w-full max-w-md">
   214→          <div className="rounded-2xl border border-vault-border bg-vault-teal/20 p-8 backdrop-blur-md">
   215→            <div className="flex flex-col items-center text-center">
   216→              <VaultMark size={44} />

   218→              <p className="mt-1 text-sm text-[#8fb0ab]">
   219→                {mode === "login" ? "Sign in to your self-assessments." : "Create your account — complete a checklist once, reuse it for a year."}

   221→            </div>

   230→                  <Label htmlFor="cp-name" className="text-verify-light/80">Full name</Label>

   243→                  className="mt-1 border-vault-border bg-vault-dark/60 text-verify-light placeholder:text-[#5c7a76]" />

   338→                  <div className="flex flex-wrap items-start justify-between gap-3">

   353→                    <div className="flex gap-2">
   354→                      <a href={`/api/checklist/pdf?completion=${c.id}`} download>
   355→                        <Button size="sm" variant="ghost" className="border border-vault-border text-[#8fb0ab] hover:text-verify-light"><Download className="mr-1.5 h-3.5 w-3.5" /> PDF</Button>
   356→                      </a>
   357→                      <Button size="sm" disabled={expired} onClick={() => setShareFor(c)} className="bg-verify-green text-vault-dark hover:bg-verify-green/90">

   366→                        {c.shareLinks.map((s) => {

   374→                              <p className="flex items-center gap-2 text-[11px]">
   375→                                <span className={cn("rounded-full border px-2 py-0.5 font-semibold", dead ? "border-vault-border text-[#5c7a76] line-through" : "border-verify-green/40 text-verify-green")}>
   376→                                  {shareAccessLabel(s)}
   377→                                </span>
   378→                                {s.accessType === "ONE_TIME" ? (s.viewedAt ? "viewed" : "unopened") : s.expiresAt ? `until ${fmt(s.expiresAt)}` : ""}
   379→                                <button type="button" onClick={() => navigator.clipboard.writeText(`${window.location.origin}/?s=${s.token}`)}
   380→                                  className="rounded p-1 text-[#8fb0ab] hover:text-verify-light" aria-label="Copy link"><Copy className="h-3 w-3" /></button>
   381→                              </p>
   382→                            </div>
   383→                          );
   384→                        })}
   385→                      </div>
   386→                    </div>
   387→                  )}
   388→                </motion.div>
   389→              );
   390→            })}
   391→          </div>
   392→        )}
   393→
   394→        {/* Requests */}
   395→        {tab === "requests" && (
   396→          <div className="mt-6 space-y-4">
   397→            {approvedRequests.map((r) => (
   398→              <div key={r.id} className="flex flex-wrap items-center justify-between gap-3 rounded-2xl border border-verify-green/30 bg-verify-green/10 p-5">
   399→                <div>
   400→                  <p className="text-sm font-semibold text-verify-light">{specialtyLabel(r.specialty)} · {r.jobTitle}</p>
   401→                  <p className="text-xs text-verify-green">Approved — ready to complete</p>
   402→                </div>
   403→                <Button size="sm" onClick={() => startFill({ requestId: r.id, profession: r.profession, jobTitle: r.jobTitle, specialty: r.specialty })}
   404→                  className="bg-verify-green text-vault-dark hover:bg-verify-green/90">Complete now</Button>
   405→              </div>
   406→            ))}
   407→            {requests.map((r) => (
   408→              <div key={r.id + r.status} className="flex flex-wrap items-center justify-between gap-3 rounded-2xl border border-vault-border bg-vault-teal/20 p-5 backdrop-blur-md">
   409→                <div>
   410→                  <p className="text-sm font-semibold text-verify-light">{specialtyLabel(r.specialty)} · {r.jobTitle} <span className="ml-1 text-xs font-normal text-[#8fb0ab]">{r.profession}</span></p>
   411→                  <p className="mt-0.5 text-xs text-[#8fb0ab]">Requested {fmt(r.requestedAt)}{r.decidedAt ? ` · decided ${fmt(r.decidedAt)}` : ""}</p>
   412→                </div>
   413→                <span className={cn("rounded-full border px-3 py-1 text-xs font-semibold",
   414→                  r.status === "PENDING" ? "border-amber-400/40 text-amber-400" : r.status === "APPROVED" ? "border-verify-green/40 text-verify-green" : "border-rose-400/40 text-rose-400")}>
   415→                  {r.status === "PENDING" ? "Awaiting superadmin approval" : r.status === "APPROVED" ? "Approved" : "Declined"}
   416→                </span>
   417→              </div>
   418→            ))}
   419→            {requests.length === 0 && (
   420→              <div className="rounded-2xl border border-dashed border-vault-border p-10 text-center">
   421→                <p className="text-sm text-[#8fb0ab]">No requests yet. Browse the library and request the checklist you need — a superadmin approves it, then it lands here.</p>
   422→                <Button size="sm" onClick={() => setShowRequest(true)} className="mt-4 bg-verify-green text-vault-dark hover:bg-verify-green/90"><Plus className="mr-1 h-3.5 w-3.5" /> Request checklist</Button>
   423→              </div>
   424→            )}
   425→          </div>
   426→        )}
   427→
   428→        {/* Invites */}
   429→        {tab === "invites" && (
   430→          <div className="mt-6 space-y-4">
   431→            {openInvites.map((i) => (
   432→              <div key={i.id} className="flex flex-wrap items-center justify-between gap-3 rounded-2xl border border-verify-green/30 bg-verify-green/10 p-5">
   433→                <div>
   434→                  <p className="text-sm font-semibold text-verify-light">
   435→                    {i.specialty ? specialtyLabel(i.specialty) : "Specialty: your choice"} <span className="text-xs font-normal text-[#8fb0ab]">· {i.jobTitle || "job title TBD"}</span>
   436→                  </p>
   437→                  <p className="mt-0.5 text-xs text-[#8fb0ab]">From {i.recruiterName}{i.facilityName ? ` · ${i.facilityName}` : ""} · {fmt(i.createdAt)}</p>
   438→                  {i.message && <p className="mt-1 text-xs italic text-[#8fb0ab]">“{i.message}”</p>}
   439→                </div>
   440→                <Button size="sm" onClick={() => startFill({ inviteId: i.id, profession: i.profession || "Nursing", jobTitle: i.jobTitle || "RN", specialty: i.specialty, recruiterName: i.recruiterName, facilityName: i.facilityName, message: i.message })}
   441→                  className="bg-verify-green text-vault-dark hover:bg-verify-green/90">Complete now</Button>
   442→              </div>
   443→            ))}
   444→            {invites.filter((i) => i.status === "COMPLETED").map((i) => (
   445→              <div key={i.id} className="flex flex-wrap items-center justify-between gap-3 rounded-2xl border border-vault-border bg-vault-teal/20 p-5 backdrop-blur-md opacity-80">
   446→                <div>
   447→                  <p className="text-sm font-semibold text-verify-light">{i.specialtyLabel || i.specialty.replace(/_/g, " ")}</p>
   448→                  <p className="mt-0.5 text-xs text-[#8fb0ab]">For {i.recruiterName} · completed {i.completedAt ? fmt(i.completedAt) : ""}</p>
   449→                </div>
   450→                <span className="flex items-center gap-1 rounded-full border border-verify-green/40 px-3 py-1 text-xs font-semibold text-verify-green"><Check className="h-3 w-3" /> Done</span>
   451→              </div>
   452→            ))}
   453→            {invites.length === 0 && (
   454→              <div className="rounded-2xl border border-dashed border-vault-border p-10 text-center">
   455→                <Inbox className="mx-auto h-8 w-8 text-verify-green/60" />
   456→                <p className="mt-3 text-sm text-[#8fb0ab]">No recruiter invites yet. When a recruiter sends you a checklist request, it shows up here.</p>
   457→              </div>
   458→            )}
   459→          </div>
   460→        )}
   461→      </main>
   462→
   463→      {/* Request-new panel */}
   464→      <AnimatePresence>
   465→        {showRequest && (
   466→          <RequestPanel sets={sets} onClose={() => setShowRequest(false)} onRequested={async () => { setShowRequest(false); setTab("requests"); await loadMe(); }} />
   467→        )}
   468→      </AnimatePresence>
   469→
   470→      {/* Share dialog */}
   471→      <AnimatePresence>
   472→        {shareFor && <ShareDialog completion={shareFor} onClose={() => setShareFor(null)} onCreated={loadMe} />}
   473→      </AnimatePresence>
   474→    </div>
   475→  );
   476→}

   478→// ── Catalog picker (candidate requests a checklist → superadmin approves) ──
   479→function RequestPanel({ sets, onClose, onRequested }: { sets: CatalogSet[]; onClose: () => void; onRequested: () => void }) {
   480→  const [busyId, setBusyId] = useState<string | null>(null);
   481→  const [error, setError] = useState<string | null>(null);
   482→
   483→  const request = async (s: CatalogSet) => {
   484→    setBusyId(s.key);
   485→    setError(null);
   486→    try {
   487→      const res = await fetch("/api/checklist/request", {
   488→        method: "POST", headers: { "Content-Type": "application/json" },
   489→        body: JSON.stringify({ profession: s.profession, jobTitle: s.jobTitle, specialty: s.specialty }),
   490→      });
   491→      const d = await res.json();
   492→      if (!res.ok || !d.ok) throw new Error(d.error ?? "Request failed");
   493→      onRequested();
   494→    } catch (e) {
   495→      setError(e instanceof Error ? e.message : "Request failed");
   496→    } finally { setBusyId(null); }
   497→  };
   498→
   499→  return (
   500→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   501→      className="fixed inset-0 z-50 flex items-end justify-center bg-vault-dark/80 p-4 backdrop-blur-sm sm:items-center" onClick={onClose}>
   502→      <motion.div initial={{ y: 24, scale: 0.98 }} animate={{ y: 0, scale: 1 }} exit={{ y: 24, opacity: 0 }} transition={{ type: "spring", stiffness: 300, damping: 26 }}
   503→        className="max-h-[80vh] w-full max-w-lg overflow-auto rounded-2xl border border-vault-border bg-[#062024] p-6" onClick={(e) => e.stopPropagation()}>
   504→        <div className="flex items-start justify-between">
   505→          <div>
   506→            <h3 className="flex items-center gap-2 text-base font-semibold text-verify-light"><Plus className="h-4 w-4 text-verify-green" /> Request a checklist</h3>
   507→            <p className="mt-1 text-xs text-[#8fb0ab]">Pick from the library. A superadmin approves your request, then you complete it once — it stays valid for a year.</p>
   508→          </div>
   509→          <button type="button" onClick={onClose} aria-label="Close" className="rounded-lg p-1.5 text-[#8fb0ab] hover:bg-white/5 hover:text-verify-light"><X className="h-4 w-4" /></button>
   510→        </div>
   511→        <div className="mt-4 space-y-2">
   512→          {sets.map((s) => (
   513→            <div key={s.key} className="flex items-center justify-between gap-3 rounded-xl border border-vault-border bg-vault-dark/40 p-4">
   514→              <div>
   515→                <p className="text-sm font-semibold text-verify-light">{s.label} <span className="text-xs font-normal text-[#8fb0ab]">· {s.jobTitle} · {s.profession}</span></p>
   516→                <p className="mt-0.5 text-xs text-[#8fb0ab]">{s.skills.length} skill(s) · {new Set(s.skills.map((k) => k.category)).size} categor{s.skills.length === 1 ? "y" : "ies"}</p>
   517→              </div>
   518→              <Button size="sm" disabled={busyId === s.key} onClick={() => request(s)} className="bg-verify-green text-vault-dark hover:bg-verify-green/90">
   519→                {busyId === s.key ? <Spinner size={14} /> : <><ExternalLink className="mr-1.5 h-3.5 w-3.5" /> Request</>}
   520→              </Button>
   521→            </div>
   522→          ))}
   523→          {sets.length === 0 && <p className="py-6 text-center text-sm text-[#8fb0ab]">The library is empty — a superadmin needs to publish checklists first.</p>}
   524→        </div>
   525→        {error && <p className="mt-3 text-xs font-medium text-rose-400">{error}</p>}
   526→      </motion.div>
   527→    </motion.div>
   528→  );
   529→}
