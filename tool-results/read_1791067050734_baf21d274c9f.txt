    73→## 5. Data contract
    74→
    75→`renderSkillsChecklistPdf(data: ChecklistData, opts?)` returns `Promise<Buffer>`. The `ChecklistData` interface is exported from the renderer; this is its shape:
    76→
    77→```json
    78→{
    79→  "candidate": { "name": "Sam", "profession": "Nursing", "jobTitle": "RN", "specialty": "General",
    80→                 "email": "sam@example.com", "experience": "1 yr(s)",
    81→                 "completedOn": "Oct 2, 2026", "validUntil": "Oct 2, 2027 (1 year)", "source": "Requested by a recruiter" },
    82→  "categories": [ { "name": "Medication Administration", "shortName": "optional, used in the Category overview",
    83→                    "skills": [ { "name": "Documentation on M.A.R.", "rating": 3, "lastPerformed": "within_6_months" } ] } ],
    84→  "questions": [ { "category": "Availability & preferences", "items": [
    85→                    { "type": "yes_no", "question": "Are you comfortable working night shifts?", "answer": true },
    86→                    { "type": "text", "question": "Which EHR systems have you used?", "answer": "Epic and Cerner." } ] } ],
    87→  "attestation": { "statements": [ { "text": "I confirm that the ratings above are my own ...", "accepted": true } ],
    88→                   "signatureName": "Sam", "printedName": "Sam", "signedOn": "Oct 2, 2026" },
    89→  "footerNote": "Self-reported assessment. Completed once by the candidate and valid for 12 months. ...",
    90→  "generatedOn": "Oct 2, 2026",
    91→  "sampleNotice": ""
    92→}
    93→```
    94→
    95→Rules:
    96→- `rating`: integer **1 to 4** (1 = No theory, 2 = Limited, 3 = Experienced, 4 = Proficient). Rating questions have **no N/A**.
    97→- `lastPerformed`: `within_3_months` | `within_6_months` | `over_6_months` | `na` (never performed).
    98→- `questions[].items[].type`: `yes_no` or `text`. **Assumed names. Confirm them against the real question-type enum and map if different.**
    99→- Dates are display strings, formatted `Oct 2, 2026`. `validUntil` is completion + 12 months, written like `Oct 2, 2027 (1 year)`.
   100→- `sampleNotice` must be **empty in production** (the approved sample sets it because its data is sample data).
   101→- Keep the order of `categories` and `skills` as in the checklist template. Do not re-sort.
   102→- If `signatureName` is empty the signature line stays blank and "Signed electronically" is hidden. An unaccepted statement renders as an empty checkbox.
   103→- Export columns (Profession, Job Title, Specialty, Category, Skill Name, Question Type, Has N/A Option): Profession, Job Title and Specialty feed the candidate chips;
   104→  Category groups the skills; Skill Name is the row label; Question Type picks the widget (`rating_1_4` = ring + chip + last-performed chip; yes/no = toggle; text = answer box).
   105→- `footerNote` / attestation statement texts: use the platform's approved legal wording. The sample wording is a draft and needs legal review.

   161→/**
   162→ * VaultVerify - Skills Checklist PDF (Node / PDFKit port)
   163→ *
   164→ * Faithful port of the approved reference implementation (generate_checklist.py).
   165→ * All coordinates are PDF points on US Letter (612 x 792) with a TOP-LEFT origin,
   166→ * which is what PDFKit uses, so the `top` values from the reference apply unchanged.
   167→ *
   168→ * Usage:
   169→ *   import { renderSkillsChecklistPdf } from "@/lib/pdf/skills-checklist-pdf";

   171→ *
   172→ * Runtime: Node.js only (never the Edge runtime).
   173→ */
   174→import path from "node:path";
   175→import PDFDocument from "pdfkit";
   176→
   177→// --------------------------------------------------------------------------- types
   178→export type Rating = 1 | 2 | 3 | 4;
   179→export type LastPerformed = "within_3_months" | "within_6_months" | "over_6_months" | "na";
   180→
   181→export interface ChecklistData {
   182→  candidate: {
   183→    name: string;
   184→    profession: string;

   195→    shortName?: string;
   196→    skills: { name: string; rating: Rating; lastPerformed: LastPerformed }[];
   197→  }[];
   198→  questions?: {
   199→    category: string;
   200→    items: (
   201→      | { type: "yes_no"; question: string; answer: boolean }
   202→      | { type: "text"; question: string; answer: string }
   203→    )[];
   204→  }[];

   257→// --------------------------------------------------------------------------- colour helpers

   267→// --------------------------------------------------------------------------- validation
   268→function validate(data: ChecklistData) {
   269→  for (const cat of data.categories) {
   270→    for (const s of cat.skills) {
   271→      if (![1, 2, 3, 4].includes(s.rating)) throw new Error(`Bad rating for "${s.name}": ${s.rating}`);

   278→// --------------------------------------------------------------------------- public API
   279→export async function renderSkillsChecklistPdf(data: ChecklistData, opts: RenderOptions = {}): Promise<Buffer> {
   280→  validate(data);
   281→  const assetsDir = opts.assetsDir ?? path.join(process.cwd(), "assets", "skills-checklist");
   282→  // Two passes: the first counts pages so "Page x of y" is right on every page.
   283→  const first = await draw(data, assetsDir, 1);
   284→  const second = await draw(data, assetsDir, first.pages);
   285→  return second.buffer;
   286→}
   287→
   288→// --------------------------------------------------------------------------- renderer
   289→function draw(data: ChecklistData, assetsDir: string, total: number): Promise<{ buffer: Buffer; pages: number }> {
   290→  const cand = data.candidate;
   291→  const cats = data.categories;
   292→  const all = cats.flatMap((c) => c.skills);
   293→  const N = all.length;
   294→  const AVG = all.reduce((a, s) => a + s.rating, 0) / N;
   295→  const PROF = all.filter((s) => s.rating === 4).length;
   296→  const RECENT = all.filter((s) => s.lastPerformed === "within_3_months").length;
   297→  const cnt = { 1: 0, 2: 0, 3: 0, 4: 0 } as Record<Rating, number>;
   298→  all.forEach((s) => cnt[s.rating]++);
   299→  const rcnt = { within_3_months: 0, within_6_months: 0, over_6_months: 0, na: 0 } as Record<LastPerformed, number>;
   300→  all.forEach((s) => rcnt[s.lastPerformed]++);
   301→  const roleLine = `${cand.name}, ${cand.jobTitle} (${cand.specialty})`;
   302→
   303→  const doc = new PDFDocument({
   304→    size: [W, H],
   305→    margin: 0,
   306→    autoFirstPage: false,
   307→    info: { Title: `Skills Checklist - ${cand.name} (${cand.jobTitle}, ${cand.specialty})`, Author: "VaultVerify" },
   308→  });
   309→  const chunks: Buffer[] = [];
   310→  doc.on("data", (c: Buffer) => chunks.push(c));
   311→
   312→  const fontDir = path.join(assetsDir, "fonts");
   313→  doc.registerFont("P", path.join(fontDir, "Poppins-Regular.ttf"));
   314→  doc.registerFont("PL", path.join(fontDir, "Poppins-Light.ttf"));
   315→  doc.registerFont("PM", path.join(fontDir, "Poppins-Medium.ttf"));
   316→  doc.registerFont("PB", path.join(fontDir, "Poppins-Bold.ttf"));
   317→  doc.registerFont("Sig", path.join(fontDir, "Lora-Italic-Regular.ttf"));
   318→  // Pass the file PATH (not a Buffer) so PDFKit embeds the logo once and reuses it on every page
   319→  const logo = path.join(assetsDir, "logo.png");
   320→
   321→  // ------------------------------------------------------------------ primitives
   322→  type Fill = string | RGB;
   323→  const rrect = (x: number, top: number, w: number, h: number, r: number, fill?: Fill, stroke?: Fill, lw = 0.8) => {
   324→    doc.roundedRect(x, top, w, h, r);
   325→    if (fill && stroke) { doc.lineWidth(lw); doc.fillAndStroke(fill as any, stroke as any); }
   326→    else if (fill) doc.fill(fill as any);
   327→    else if (stroke) { doc.lineWidth(lw); doc.stroke(stroke as any); }
   328→  };
   329→  const width = (s: string, font = "P", size = 9) => doc.font(font).fontSize(size).widthOfString(s);
   330→  const text = (x: number, top: number, s: string, font = "P", size = 9, color: Fill = INK, align: "l" | "r" | "c" = "l") => {
   331→    doc.font(font).fontSize(size).fillColor(color as any);
   332→    const w = doc.widthOfString(s);
   333→    const x0 = align === "l" ? x : align === "r" ? x - w : x - w / 2;
   334→    doc.text(s, x0, top, { lineBreak: false, baseline: "alphabetic" });
   335→  };
   336→  const line = (x1: number, y1: number, x2: number, y2: number, color: Fill, lw: number) => {
   337→    doc.lineWidth(lw).strokeColor(color as any).moveTo(x1, y1).lineTo(x2, y2).stroke();
   338→  };
   339→  /** ReportLab simpleSplit: greedy word wrap on spaces. */
   340→  const wrap = (s: string, font: string, size: number, maxW: number): string[] => {
   341→    const out: string[] = [];
   342→    for (const para of s.split("\n")) {
   343→      let cur = "";
   344→      for (const w of para.split(/\s+/).filter(Boolean)) {
   345→        const cand2 = cur ? `${cur} ${w}` : w;
   346→        if (!cur || width(cand2, font, size) <= maxW) cur = cand2;
   347→        else { out.push(cur); cur = w; }
   348→      }
   349→      out.push(cur);
   350→    }
   351→    return out;
   352→  };
   353→  const alpha = (fn: () => void, a: number) => { doc.save(); doc.fillOpacity(a); fn(); doc.restore(); };
   354→
   355→  const headerRings = (hb: number) => {
   356→    doc.save();
   357→    doc.rect(0, 0, W, hb).clip();
   358→    for (const [r, a] of [[150, 0.06], [112, 0.07], [74, 0.08]] as const) {
   359→      doc.fillColor([77, 171, 87] as any).fillOpacity(a).circle(W - 40, 10, r).fill();
   360→    }
   361→    doc.restore();
   362→  };
   363→
   364→  // ------------------------------------------------------------------ rating widgets
   365→  /** 4-segment score ring; segments fill clockwise from 12 o'clock up to `level`, number centred. */

   744→export async function GET(req: Request, ctx: { params: Promise<{ id: string }> }) {
   745→  // Next.js 15+: params is a Promise. On Next 13/14 use `{ params }: { params: { id: string } }` and drop the await.
   746→  const { id } = await ctx.params;
   747→
   748→  if (!(await isAuthorised(req, id))) {
   749→    // 404 (not 403) so the response does not reveal whether a checklist exists
   750→    return new Response("Not found", { status: 404 });
   751→  }
   752→  const data = await loadChecklistData(id);
   753→  if (!data) return new Response("Not found", { status: 404 });
   754→
   755→  const pdf = await renderSkillsChecklistPdf(data);
   756→  return new Response(new Uint8Array(pdf), {
   757→    headers: {
   758→      "Content-Type": "application/pdf",
   759→      "Content-Disposition": `inline; filename="skills-checklist-${id}.pdf"`,
   760→      "Cache-Control": "private, no-store",
   761→    },
   762→  });
   763→}
   764→```

   804→### `scripts/compare-pdfs.py` (golden comparison)
   805→
   806→```python
   807→"""Golden comparison: python compare-pdfs.py approved-design.pdf candidate.pdf
   808→Rasterises both at 100 dpi and reports differing pixels per page. Needs: poppler (pdftoppm), pillow, numpy."""
   809→import subprocess, sys, tempfile, os
   810→from PIL import Image
   811→import numpy as np
   812→a, b = sys.argv[1], sys.argv[2]
   813→with tempfile.TemporaryDirectory() as d:
   814→    for tag, f in (("a", a), ("b", b)):
   815→        subprocess.run(["pdftoppm", "-r", "100", "-png", f, os.path.join(d, tag)], check=True)
   816→    pa = sorted(x for x in os.listdir(d) if x.startswith("a"))
   817→    pb = sorted(x for x in os.listdir(d) if x.startswith("b"))
   818→    print("pages:", len(pa), "vs", len(pb))
   819→    ok = len(pa) == len(pb)
   820→    for x, y in zip(pa, pb):
   821→        ia = np.asarray(Image.open(os.path.join(d, x)).convert("RGB")).astype(int)
   822→        ib = np.asarray(Image.open(os.path.join(d, y)).convert("RGB")).astype(int)
   823→        if ia.shape != ib.shape:
   824→            print(x, "size mismatch"); ok = False; continue
   825→        n = int((np.abs(ia - ib).max(axis=2) > 40).sum())
   826→        print(x, "differing pixels:", n)
   827→        ok &= n < 300   # tolerance: a few hairline/dash pixels
   828→    print("PASS" if ok else "FAIL"); sys.exit(0 if ok else 1)
   829→```
